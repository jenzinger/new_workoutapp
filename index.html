/* Hardened Storage Function for Desktop & Mobile */ function saveAppState() { const state = { prepDuration, workDuration, restDuration, totalCycles, exercises }; try { window.localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); } catch(e) { console.error("Failed to save state to localStorage:", e); } } function loadAppState() { try { const saved = window.localStorage.getItem(STORAGE_KEY); if (saved) { const state = JSON.parse(saved); if (state.prepDuration !== undefined) prepDuration = Number(state.prepDuration); if (state.workDuration !== undefined) workDuration = Number(state.workDuration); if (state.restDuration !== undefined) restDuration = Number(state.restDuration); if (state.totalCycles !== undefined) totalCycles = Number(state.totalCycles); if (Array.isArray(state.exercises) && state.exercises.length > 0) { exercises = state.exercises; } } } catch(e) { console.error("Failed to load state from localStorage:", e); } document.getElementById('prepInput').value = prepDuration; document.getElementById('workInput').value = workDuration; document.getElementById('restInput').value = restDuration; document.getElementById('cyclesInput').value = totalCycles; } // Immediately flush on every input change, keyup, or focus out on PC document.addEventListener('DOMContentLoaded', () => { const inputs = ['prepInput', 'workInput', 'restInput', 'cyclesInput']; inputs.forEach(id => { const el = document.getElementById(id); if (el) { el.addEventListener('change', updateSettings); el.addEventListener('keyup', updateSettings); el.addEventListener('blur', updateSettings); } }); }); window.addEventListener('beforeunload', saveAppState); window.addEventListener('pagehide', saveAppState);
